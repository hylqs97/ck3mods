autoedu.0106 = {
  type = character_event
  hidden = yes
  trigger = {
    is_ai = no
    has_character_flag = autoedu_ward_reminder_scan_loop_started
    autoedu_release_scheduler_schema_compatible_trigger = yes
  }
  immediate = {
    if = { limit = { exists = var:autoedu_settings_initialized NOT = { exists = var:autoedu_editor_schema_version } } autoedu_release_migrate_legacy_1_10_effect = yes }
    autoedu_release_prepare_context_effect = yes
    if = {
      limit = { autoedu_editor_current_schema_valid_trigger = yes has_character_flag = autoedu_ward_reminder_scan_loop_started }

        limit = {
          has_bp2_dlc_trigger = yes
          var:autoedu_editor_general_personality_mode = 1
          NOT = { exists = var:autoedu_ward_reminder_personality_queued }
        }
        autoedu_find_personality_influence_ward_for_owner_effect = yes
      }
      if = {
        limit = {
          has_mpo_dlc_trigger = yes
          var:autoedu_editor_general_martial_training_mode = 1
          NOT = { exists = var:autoedu_ward_reminder_martial_queued }
        }
        autoedu_find_martial_training_ward_for_owner_effect = yes
      }
      if = {
        limit = {
          exists = scope:autoedu_personality_child
          exists = scope:autoedu_martial_training_child
        }
        set_variable = { name = autoedu_ward_reminder_personality_age value = scope:autoedu_personality_child.age }
        set_variable = { name = autoedu_ward_reminder_martial_age value = scope:autoedu_martial_training_child.age }
        if = {
          limit = { var:autoedu_ward_reminder_personality_age >= var:autoedu_ward_reminder_martial_age }
          autoedu_enqueue_ward_reminder_effect = { TYPE = 1 CHILD = scope:autoedu_personality_child QUEUED_VAR = autoedu_ward_reminder_personality_queued }
          autoedu_enqueue_ward_reminder_effect = { TYPE = 2 CHILD = scope:autoedu_martial_training_child QUEUED_VAR = autoedu_ward_reminder_martial_queued }
        }
        else = {
          autoedu_enqueue_ward_reminder_effect = { TYPE = 2 CHILD = scope:autoedu_martial_training_child QUEUED_VAR = autoedu_ward_reminder_martial_queued }
          autoedu_enqueue_ward_reminder_effect = { TYPE = 1 CHILD = scope:autoedu_personality_child QUEUED_VAR = autoedu_ward_reminder_personality_queued }
        }
        remove_variable = autoedu_ward_reminder_personality_age
        remove_variable = autoedu_ward_reminder_martial_age
      }
      else_if = {
        limit = { exists = scope:autoedu_personality_child }
        autoedu_enqueue_ward_reminder_effect = { TYPE = 1 CHILD = scope:autoedu_personality_child QUEUED_VAR = autoedu_ward_reminder_personality_queued }
