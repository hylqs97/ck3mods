open_advanced_vassal_manager_menu = {
    picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	decision_group_type = courtier
    desc = open_advanced_vassal_manager_menu_desc
    selection_tooltip = open_advanced_vassal_manager_menu_desc
    confirm_text = CONFIRM_OKAY

    ai_check_interval = 0

    is_shown = {
        is_ruler = yes
		is_ai = no
	    NOT = { has_character_flag = advanced_vassal_manager }
    }

    effect = {
        add_character_flag = {
            flag = advanced_vassal_manager
        }
    }
	
    ai_potential = {
		always = no
	}
	ai_will_do = {
		base = 0
	}
}
