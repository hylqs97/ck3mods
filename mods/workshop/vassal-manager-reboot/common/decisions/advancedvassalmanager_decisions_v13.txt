# ---------------------------------------------------------------------
# SAFETY: arm / disarm the three unrecoverable revoke decisions.
# Arming lasts 30 days and then lapses on its own.
# ---------------------------------------------------------------------

avm_arm_mass_revocation_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	decision_group_type = courtier
	desc = avm_arm_mass_revocation_decision_desc
	selection_tooltip = avm_arm_mass_revocation_decision_tooltip
	confirm_text = CONFIRM_OKAY

	ai_check_interval = 0

	is_shown = {
		is_ruler = yes
		is_ai = no
		has_character_flag = advanced_vassal_manager
		has_character_flag = avm_revoke_titles
		NOT = { has_character_flag = avm_armed }
	}

	effect = {
		add_character_flag = {
			flag = avm_armed
			days = 30
		}
	}

	ai_potential = { always = no }
	ai_will_do = { base = 0 }
}

# ---------------------------------------------------------------------
# Revoke from vassals who actively dislike you.
# This is the filter the pinned-character system was a manual stand-in for.
# ---------------------------------------------------------------------

revoke_disloyal_vassal_titles_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	decision_group_type = courtier
	desc = revoke_disloyal_vassal_titles_decision_desc
	selection_tooltip = revoke_disloyal_vassal_titles_decision_tooltip
	confirm_text = CONFIRM_OKAY

	ai_check_interval = 0

	is_shown = {
		is_ruler = yes
		is_ai = no
		has_character_flag = advanced_vassal_manager
		has_character_flag = avm_revoke_titles
	}

	is_valid = {
		any_sub_realm_title = {
			tier <= tier_empire
				# v1.4: never touch a protected character
				NOT = { holder = { has_character_flag = avm_protected } }
			is_titular = no
			holder = {
				trigger_if = {
					limit = { has_game_rule = avm_disloyalty_lenient }
					opinion = { target = root  value <= -10 }
				}
				trigger_else_if = {
					limit = { has_game_rule = avm_disloyalty_strict }
					opinion = { target = root  value <= -50 }
				}
				trigger_else = {
					opinion = { target = root  value <= -25 }
				}
			}
			count >= 1
		}
	}

	cooldown = { days = 1 }

	effect = {
		# v1.4: this operation replaces any earlier undo record
		clear_variable_list = avm_undo_titles
		every_sub_realm_title = {
			limit = {
				tier <= tier_empire
				is_capital_barony = no
				# v1.4: never touch a protected character
				NOT = { holder = { has_character_flag = avm_protected } }
				is_titular = no
				holder = {
					# v1.4: threshold comes from the AVM game rule
					trigger_if = {
						limit = { has_game_rule = avm_disloyalty_lenient }
						opinion = { target = root  value <= -10 }
					}
					trigger_else_if = {
						limit = { has_game_rule = avm_disloyalty_strict }
						opinion = { target = root  value <= -50 }
					}
					trigger_else = {
						opinion = { target = root  value <= -25 }
					}
				}
			}
			add_to_temporary_list = target_title_disloyal
		}
		every_in_list = {
			list = target_title_disloyal
			limit = {
				holder = {
					target_is_liege_or_above = root
				}
			}
			# v1.4: remember the previous holder so Restore can put it back
			save_scope_as = avm_undo_title
			set_variable = {
				name = avm_prev_holder
				value = this.holder
			}
			root = {
				add_to_variable_list = {
					name = avm_undo_titles
					target = scope:avm_undo_title
				}
			}
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = change
				add_claim_on_loss = no
			}
			change_title_holder = {
				holder = root
				change = scope:change
			}
			resolve_title_and_vassal_change = scope:change
		}
	}

	ai_potential = { always = no }
	ai_will_do = { base = 0 }
}


# ---------------------------------------------------------------------
# Government conversion. The original mod only ever converted HOLDINGS,
# and explicitly skipped anything held by a tribal-government character,
# so tribal vassals stayed tribal forever. These change the government
# first, then bring the holdings into line with it.
# ---------------------------------------------------------------------

convert_tribal_vassals_feudal_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	decision_group_type = courtier
	desc = convert_tribal_vassals_feudal_decision_desc
	selection_tooltip = convert_tribal_vassals_feudal_decision_tooltip
	confirm_text = CONFIRM_OKAY

	ai_check_interval = 0

	is_shown = {
		is_ruler = yes
		is_ai = no
		has_character_flag = advanced_vassal_manager
		has_character_flag = avm_misc
		NOT = { government_has_flag = government_is_tribal }
	}

	is_valid = {
		any_vassal_or_below = {
			government_has_flag = government_is_tribal
			count >= 1
		}
	}

	cooldown = { days = 1 }

	effect = {
		every_vassal_or_below = {
			limit = {
				government_has_flag = government_is_tribal
				# v1.4: optionally respect the innovation vanilla requires
				trigger_if = {
					limit = { has_game_rule = avm_conversion_respect_innovations }
					culture = { has_innovation = innovation_manorialism }
				}
			}
			add_to_temporary_list = target_tribal_subjects
		}
		every_in_list = {
			list = target_tribal_subjects
			change_government = feudal_government
			every_held_title = {
				limit = {
					tier <= tier_county
					is_capital_barony = no
					title_province = {
						has_holding_type = tribal_holding
					}
				}
				title_province = {
					set_holding_type = castle_holding
				}
			}
		}
	}

	ai_potential = { always = no }
	ai_will_do = { base = 0 }
}
