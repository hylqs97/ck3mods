# ---------------------------------------------------------------------
# Clan -> feudal. No holding conversion step: clan rulers already hold
# castles, so only the government changes.
# ---------------------------------------------------------------------

convert_clan_vassals_feudal_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	decision_group_type = courtier
	desc = convert_clan_vassals_feudal_decision_desc
	selection_tooltip = convert_clan_vassals_feudal_decision_tooltip
	confirm_text = CONFIRM_OKAY

	ai_check_interval = 0

	is_shown = {
		is_ruler = yes
		is_ai = no
		has_character_flag = advanced_vassal_manager
		has_character_flag = avm_misc
		NOT = { government_has_flag = government_is_tribal }
	}

	is_valid = {
		any_vassal_or_below = {
			government_has_flag = government_is_clan
			count >= 1
		}
	}

	cooldown = { days = 1 }

	effect = {
		every_vassal_or_below = {
			limit = {
				government_has_flag = government_is_clan
				NOT = { has_character_flag = avm_protected }
			}
			add_to_temporary_list = target_clan_subjects
		}
		every_in_list = {
			list = target_clan_subjects
			change_government = feudal_government
		}
	}

	ai_potential = { always = no }
	ai_will_do = { base = 0 }
}


# ---------------------------------------------------------------------
# Culture equivalent of Convert Subjects' Faith. Characters only by
# default; the game rule adds county culture on top.
# ---------------------------------------------------------------------

convert_subjects_culture_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	decision_group_type = courtier
	desc = convert_subjects_culture_decision_desc
	selection_tooltip = convert_subjects_culture_decision_tooltip
	confirm_text = CONFIRM_OKAY

	ai_check_interval = 0

	is_shown = {
		is_ruler = yes
		is_ai = no
		has_character_flag = advanced_vassal_manager
		has_character_flag = avm_misc
	}

	cooldown = { days = 1 }

	effect = {
		# v1.5: courtiers are not vassals, so they need their own pass
		every_vassal_or_below = {
			limit = {
				NOT = { has_character_flag = avm_protected }
			}
			set_culture = root.culture
			every_courtier = {
				limit = {
					NOT = { has_character_flag = avm_protected }
				}
				set_culture = root.culture
			}
		}
		every_courtier = {
			limit = {
				NOT = { has_character_flag = avm_protected }
			}
			set_culture = root.culture
		}
	}

	ai_potential = { always = no }
	ai_will_do = { base = 0 }
}


# ---------------------------------------------------------------------
# Undo. Every revoke decision records who held each title before it was
# taken. This puts them back. Only the most recent revoke is covered -
# each one clears the previous record.
# ---------------------------------------------------------------------

avm_restore_previous_holders_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	decision_group_type = courtier
	desc = avm_restore_previous_holders_decision_desc
	selection_tooltip = avm_restore_previous_holders_decision_tooltip
	confirm_text = CONFIRM_OKAY

	ai_check_interval = 0

	is_shown = {
		is_ruler = yes
		is_ai = no
		has_character_flag = advanced_vassal_manager
		has_character_flag = avm_revoke_titles
	}

	is_valid = {
		any_in_list = {
			variable = avm_undo_titles
			count >= 1
		}
	}

	effect = {
		every_in_list = {
			variable = avm_undo_titles
			limit = {
				holder = root
				exists = var:avm_prev_holder
				var:avm_prev_holder = {
					is_alive = yes
				}
			}
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = change
				add_claim_on_loss = no
			}
			change_title_holder = {
				holder = var:avm_prev_holder
				change = scope:change
			}
			resolve_title_and_vassal_change = scope:change
			remove_variable = avm_prev_holder
		}
		clear_variable_list = avm_undo_titles
	}

	ai_potential = { always = no }
	ai_will_do = { base = 0 }
}

# ---------------------------------------------------------------------
# Succession law. Ignores the crown authority and innovation requirements
# vanilla would enforce.
#
# NOTE: succession is a REALM law on the character, not a title law. The
# effect is add_realm_law_skip_effects and the law is
# single_heir_succession_law - "primogeniture" is the display name, not the
# key. Adding a law a character already has logs an error, hence the guard.
# ---------------------------------------------------------------------

set_vassals_primogeniture_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	decision_group_type = courtier
	desc = set_vassals_primogeniture_decision_desc
	selection_tooltip = set_vassals_primogeniture_decision_tooltip
	confirm_text = CONFIRM_OKAY

	ai_check_interval = 0

	is_shown = {
		is_ruler = yes
		is_ai = no
		has_character_flag = advanced_vassal_manager
		has_character_flag = avm_misc
	}

	is_valid = {
		any_vassal_or_below = {
			NOT = { has_realm_law = single_heir_succession_law }
			count >= 1
		}
	}

	cooldown = { days = 1 }

	effect = {
		every_vassal_or_below = {
			limit = {
				NOT = { has_character_flag = avm_protected }
				# v1.7.1: adding a law a character already has throws an error
				NOT = { has_realm_law = single_heir_succession_law }
			}
			add_realm_law_skip_effects = single_heir_succession_law
		}
	}

	ai_potential = { always = no }
	ai_will_do = { base = 0 }
}

# Spawn identity. Controls the culture and faith of characters created by

# county being granted. "Chosen" copies whichever county you are sitting in

		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"

# ---------------------------------------------------------------------
# Revoke by government type. Same shape as the original revoke decisions,
# filtering on the holder's government instead of faith or culture.
# ---------------------------------------------------------------------

revoke_tribal_gov_titles_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	decision_group_type = courtier
	desc = revoke_tribal_gov_titles_decision_desc
	selection_tooltip = revoke_tribal_gov_titles_decision_tooltip
	confirm_text = CONFIRM_OKAY

	ai_check_interval = 0

	is_shown = {
		is_ruler = yes
		is_ai = no
		has_character_flag = advanced_vassal_manager
		has_character_flag = avm_revoke_titles
	}

	is_valid = {
		any_sub_realm_title = {
			tier <= tier_empire
			is_capital_barony = no
			is_titular = no
			NOT = { holder = { has_character_flag = avm_protected } }
			holder = { government_has_flag = government_is_tribal }
			count >= 1
		}
	}

	cooldown = { days = 1 }

	effect = {
		# v1.9: revoke from tribal-government holders
		clear_variable_list = avm_undo_titles
		every_sub_realm_title = {
			limit = {
				tier <= tier_empire
				is_capital_barony = no
				is_titular = no
				NOT = { holder = { has_character_flag = avm_protected } }
			holder = { government_has_flag = government_is_tribal }
			}
			add_to_temporary_list = avm_gov_targets
		}
		every_in_list = {
			list = avm_gov_targets
			limit = {
				holder = { target_is_liege_or_above = root }
			}
			save_scope_as = avm_undo_title
			set_variable = {
				name = avm_prev_holder
				value = this.holder
			}
			root = {
				add_to_variable_list = {
					name = avm_undo_titles
					target = scope:avm_undo_title
				}
			}
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = change
				add_claim_on_loss = no
			}
			change_title_holder = {
				holder = root
				change = scope:change
			}
			resolve_title_and_vassal_change = scope:change
		}
	}

	ai_potential = { always = no }
	ai_will_do = { base = 0 }
}

		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}

	selection_tooltip = avm_grant_down_to_domain_limit_decision_tooltip

		is_ai = no
		has_character_flag = advanced_vassal_manager
		has_character_flag = avm_grant_titles
	}

	is_valid = {
		domain_size > domain_limit
	}

	cooldown = { days = 1 }

		while = {
			limit = {
				domain_size > domain_limit
				any_held_title = {

				create_title_and_vassal_change = {
					type = granted
					save_scope_as = change
					add_claim_on_loss = no
				}
				change_title_holder = {
					holder = scope:receiver
					change = scope:change
				}
				resolve_title_and_vassal_change = scope:change
			}
		}
	}
