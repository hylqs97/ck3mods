# Script values - v1.7
# The cost of the funding decisions scales with the number of landed vassals
# who will actually receive gold. The limit here MUST match the limit in the
# decision effect, or you will pay for vassals who get nothing.

avm_funding_cost_modest = {
	value = 0
	every_vassal = {
		limit = {
			is_landed = yes
			NOT = { has_character_flag = avm_protected }
		}
		add = 50
	}
}

# --- v2.2: counts, so decisions can say how much they will do -------------
# Each of these MUST mirror the limit in the matching decision effect.

avm_count_all_titles = {
	value = 0
	every_sub_realm_title = {
		limit = {
			tier <= tier_empire
			is_capital_barony = no
			is_titular = no
			NOT = { holder = { has_character_flag = avm_protected } }
		}
		add = 1
	}
}

avm_count_infidel_titles = {

			NOT = { holder = { has_character_flag = avm_protected } }

}
