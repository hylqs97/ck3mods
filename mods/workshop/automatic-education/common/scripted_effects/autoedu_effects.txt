autoedu_assign_guardian_effect = {
  $GUARDIAN$ = {
    save_scope_as = autoedu_guardian
  }

  if = {
    limit = {
      exists = scope:autoedu_child
      scope:autoedu_child = {
        is_alive = yes
        is_hostage = no
        is_adult = no
        num_of_relation_guardian = 0
        NOT = { has_character_flag = autoedu_excluded_from_automatic_education }
        NOT = { has_character_flag = autoedu_guardian_assignment_recently_done }
      }
    }

    scope:autoedu_child = {
      add_character_flag = {
        flag = autoedu_guardian_assignment_recently_done
        days = 14
      }

      set_relation_guardian = $GUARDIAN$

      scope:autoedu_owner = {
        autoedu_apply_guardian_conversion_flags_effect = { GUARDIAN = $GUARDIAN$ }
      }

      if = {
        limit = {
          $GUARDIAN$ = { is_landed = yes }
          is_landed = no
          is_hostage = no
          NOT = { $GUARDIAN$ = host }
        }
        visit_court_of = $GUARDIAN$
      }
      else_if = {
        limit = {
          $GUARDIAN$ = {
            is_landed = no
            is_hostage = no
          }
          is_landed = yes
          NOT = { host = $GUARDIAN$ }
        }
        $GUARDIAN$ = {
          visit_court_of = scope:autoedu_child
        }
      }
      else_if = {
        limit = {
          $GUARDIAN$ = { is_landed = no }
          is_landed = no
          is_hostage = no
          exists = $GUARDIAN$.host
          NOT = { $GUARDIAN$.host = host }
        }
        visit_court_of = $GUARDIAN$.host
      }
    }

    if = {
      limit = { scope:autoedu_owner.var:autoedu_notifications = yes }
      scope:autoedu_owner = {
        send_interface_message = {
          type = msg_autoedu_good
          title = autoedu_guardian_assigned_title
          desc = autoedu_guardian_assigned_desc
          left_icon = scope:autoedu_child
          right_icon = scope:autoedu_guardian
        }
      }
    }
  }

  scope:autoedu_owner = { autoedu_finish_guardian_loss_child_effect = yes }
}
