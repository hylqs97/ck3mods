# On Actions for Automated Courtier Management

yearly_playable_pulse = {
	on_actions = {
		auto_court_yearly_pulse_on_action
	}
}

auto_court_yearly_pulse_on_action = {
	trigger = {
		is_ai = no
		is_ruler = yes
		NOT = { has_variable = auto_court_disabled }
	}
	effect = {
		auto_court_run_yearly_management_effect = yes
	}
}
