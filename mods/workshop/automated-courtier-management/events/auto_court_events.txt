namespace = auto_court_event

# -------------------------------------------------------------------
# Event 0001: Main Configuration Dashboard
# -------------------------------------------------------------------
auto_court_event.0001 = {
	type = character_event
	title = auto_court_event.0001.t
	desc = auto_court_event.0001.desc
	theme = stewardship_duty_focus
	override_background = { reference = council_chamber }

	right_portrait = {
		character = root
		animation = happy_teacher
		camera = camera_body_left_slight
	}
	
	immediate = {
		auto_court_initialize_settings_effect = yes
	}

	# Option: Toggle Main Automation On/Off
	option = {
		name = auto_court_event.0001.toggle_status
		if = {
			limit = {
				has_variable = auto_court_enabled
			}
			remove_variable = auto_court_enabled
			set_variable = {
				name = auto_court_disabled
				value = 1
			}
		}
		else = {
			remove_variable = auto_court_disabled
			set_variable = {
				name = auto_court_enabled
				value = 1
			}
		}
		trigger_event = {
			id = auto_court_event.0001
		}
	}

	# Option: Configuration Submenu
	option = {
		name = auto_court_event.0001.configuration_menu
		trigger_event = {
			id = auto_court_event.0006
		}
	}

	# Option: Run Evaluation Now
	option = {
		name = auto_court_event.0001.run_now
		auto_court_run_yearly_management_effect = yes
	}

	# Debug Option (Commented out for production)
	# option = {
	# 	name = auto_court_event.0001.debug_spawn
	# 	auto_court_debug_spawn_test_courtiers_effect = yes
	# 	trigger_event = {
	# 		id = auto_court_event.0001
	# 	}
	# }

	# Option: Finish & Close
	option = {
		name = auto_court_event.0001.close
	}
}
